% Verification witness tags — the family grammar (contract: 05-kkgt
% docs/witness-contract.md §6, linked from docs/style-guide.md). ONE macro,
% ONE argument order, two placements:
%   \witness{X}        -> [X]     the witness sits in the prose of the SAME
%                                sentence (a \ref/\eqref of this manuscript,
%                                the name of one of our Lean declarations, or
%                                a block of its own).  A site with no ids
%                                stays class-pure: the bracket holds the
%                                class and no id is invented for it.
%   \witness[ids]{X}   -> X[ids]  the optional argument holds INTERNAL ids
%                                only — a \label key of this manuscript or a
%                                raw Lean declaration name, several of them
%                                separated by ',', printed in tt inside the
%                                tag's own brackets.  An external source is
%                                never a key of the tag: it is a real \cite
%                                set next to the tag ([N]~\cite{key},
%                                [P] in~\cite{key}), so a tag never prints a
%                                number in its bracket.  A \cite in the
%                                optional argument raises \PackageError.
%                                The argument itself is read as an
%                                ARBITRARY STRING (contract §6.5): \_ and a
%                                raw _ both print as the breakable
%                                typewriter underscore and every other token
%                                passes through, so no valid id is rejected
%                                by its shape.  Whether a bracket holds a
%                                citation key is measured where the key set
%                                exists, by the suite against
%                                citation.csl.json.
% The placement (inline vs own line) is decided by the quantified criterion of
% docs/witness-contract.md §6.10 (k >= 2, L >= 35, r = 0, i = 0, d >= 75,
% amended by the heuristics u and p); the applied inventory lives in
% docs/reports/20261116_WIT-001_criterion_sites.md.
\makeatletter
\@namedef{markcol@P}{\color{tagP}}   % blue  — formal proof (Lean)
\@namedef{markcol@N}{\color{tagN}}   % green — numerical check
\@namedef{markcol@H}{\color{tagH}}   % red   — hypothesis; no H mark is printed
% today (the class was removed, WIT-001 §4), but the key exists so that an
% H mark can never fall through to black silently (§6.12 S6)
% unknown key: no colour, no error (the 05-kkgt form of \markcol)
\newcommand{\markcol}[1]{\expandafter\ifx\csname markcol@#1\endcsname\relax\else\csname markcol@#1\endcsname\fi}
% The one constant the tags print an underscore through (contract §6.12 S8):
% line breaking of an id list is comma-first, "_" only as a fallback.  The
% break after ", " is the natural interword break (penalty 0) and the break
% inside an identifier is DISCOURAGED (penalty 500), so a list splits at its
% commas and only an identifier longer than the remaining measure is split at
% all -- never \allowbreak (= penalty 0), which makes the mid-identifier break
% equally attractive, and never a bare \textunderscore (the old form of this
% manuscript), which is the opposite extreme: no break inside an id at all,
% the one that can go overfull.  The penalty-tuned form sits between them.
% The trailing \relax is load-bearing: after a <dimen> TeX scans the keywords
% `plus' and `minus' of \hskip, and an identifier that begins -- right after
% the underscore -- with the letters `minus' or `plus' would be eaten as that
% keyword ("Missing number ... Illegal unit of measure").  \relax is not a
% letter, so both keyword scans push it back and the id prints whole.
\DeclareRobustCommand{\brk@us}{\textunderscore\penalty500\hskip0pt\relax}
\ExplSyntaxOn
\tl_new:N \l__tier_src_tl
\tl_new:N \l__tier_us_tl   \tl_set:Nx \l__tier_us_tl  { \char_generate:nn { 95 } { 8  } }
\tl_new:N \l__tier_usx_tl  \tl_set:Nx \l__tier_usx_tl { \char_generate:nn { 95 } { 11 } }
\tl_new:N \l__tier_uso_tl  \tl_set:Nx \l__tier_uso_tl { \char_generate:nn { 95 } { 12 } }
% \tierids{<arbitrary string>}: \_ and a raw _ (any category code) both
% become the breakable typewriter underscore; every other token of the
% argument passes through unchanged.  \tl_replace_all does not descend
% into brace groups, so every wrapper applies the normalisation to its own
% argument.
\cs_new_protected:Npn \tierids #1
  {
    \tl_set:Nn \l__tier_src_tl {#1}
    \tl_replace_all:Nnn \l__tier_src_tl { \_ } { \brk@us }
    \exp_args:NNV \tl_replace_all:Nnn \l__tier_src_tl \l__tier_us_tl  { \brk@us }
    \exp_args:NNV \tl_replace_all:Nnn \l__tier_src_tl \l__tier_usx_tl { \brk@us }
    \exp_args:NNV \tl_replace_all:Nnn \l__tier_src_tl \l__tier_uso_tl { \brk@us }
    \tl_use:N \l__tier_src_tl
  }
\ExplSyntaxOff
% The character-shape grammar of the contract (docs/witness-contract.md
% §6.4: a \testref, or raw names with \_ or ',', anything else an error) is
% NOT installed here, and no longer needs to be: it rejected the ids this
% body still carries (18 id parts — 12 Lean declaration names of
% lean/CW6_complete_v2.lean and 6 ellipsis placeholders in gloss
% sentences), which remain work items to rewrite in snake_case.
% The argument is read as an arbitrary string instead, and the
% citation-key rule is measured by the suite, where the key set exists; the
% \cite guard below is the check this manuscript runs at run time.
% THE MARK -- ONE name, placement-driven (contract §6.12 S7): the shape
% follows the position of the call in the source, never the class.
%   own paragraph (\ifvmode)  -> the block, flush right, \medskip above and
%                               below, indent 0, no terminal punctuation
%                               (S2-S5);
%   mid-line in the source
%   (\ifhmode)               -> the mark inside the sentence: no \par, no
%                               indent, no space is eaten, the paragraph
%                               does not break.
% ONE argument order for both branches: \witness[ids]{X} — the ids first,
% in the optional argument, the class second; a mark with no ids prints the
% class-pure bracket [X] (contract §6.1).  The body selects its own font
% (\normalfont ... \mdseries\upshape), so an enclosing \itshape (amsthm's
% plain theorem style) cannot print the mark bold italic; colour is ONE
% branch for the whole mark (S6), the letter and the ids in the class
% colour; the sizes are the family rule 13/15 (CVR-001 TYP-4): the letter
% in \footnotesize, the ids in \small tt, upright.
% \witness@text keeps a mark written inside $...$ a TEXT mark (contract
% §6.12 S7).  It replaces the \mbox the inline macro wrapped its mark in: a
% \mbox freezes a long id list and defeats the S8 break policy.
\def\witness@text#1{\ifmmode\text{#1}\else#1\fi}
\def\witness@mark#1#2{%
  \ifx\relax#1\relax
    {\normalfont\footnotesize\bfseries\markcol{#2}[#2]}%
  \else
    {\normalfont\footnotesize\bfseries\markcol{#2}#2{\mdseries\upshape\small\ttfamily[\tierids{#1}]}}%
  \fi}
\newcommand{\witness}[2][]{%
  \ifvmode
    \par\medskip\noindent\hfill
    \parbox{\dimexpr\linewidth-1em\relax}{\raggedleft\witness@mark{#1}{#2}}%
    \par\medskip
  \else
    \in@{\cite}{#1}\ifin@
      \PackageError{witness}{the optional argument of \string\witness\space holds internal ids only}
      {An external source is a \string\cite\space set next to the tag, never its key;}%
    \else
      \witness@text{\witness@mark{#1}{#2}}%
    \fi
  \fi}
